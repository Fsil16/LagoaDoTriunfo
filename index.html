async function carregarDashboard() { const barraGeral = document.getElementById('barra-geral'); try { // Usamos apenas a URL pura. O Google Apps Script não lida bem com // cabeçalhos complexos vindos de domínios externos como o GitHub. const resposta = await fetch(urlAPI, { method: 'GET', mode: 'no-cors', // TENTATIVA 1: Se o seu script apenas envia dados, o no-cors pode liberar a conexão redirect: 'follow' }); // NOTA: No modo 'no-cors', o JS não consegue ler o JSON por segurança. // Se o erro persistir, volte para o modo padrão abaixo, que é o mais correto: const respostaReal = await fetch(urlAPI + "?t=" + new Date().getTime(), { method: 'GET', redirect: 'follow' }); const dados = await respostaReal.json(); // Atualização dos campos document.getElementById('data-inicio').innerText = "📅 INÍCIO: " + formatarData(dados.info.inicio); document.getElementById('data-atualizacao').innerText = "🔄 ATUALIZAÇÃO: " + formatarData(dados.info.atualizacao); barraGeral.innerText = `Produção: ${dados.info.operacao || 0} Metros`; const containerRetiros = document.getElementById('lista-retiros'); if (dados.retiros && dados.retiros.length > 0) { let html = ""; dados.retiros.forEach(r => { const p = parseFloat(r.perc) || 0; html += `
${r.nome} ${r.executado} / ${r.total} m
${p}%
`; }); containerRetiros.innerHTML = html; } } catch (erro) { console.error("Erro detectado:", erro); barraGeral.innerText = "Erro: Verifique a conexão ou publicação do Script"; } }
