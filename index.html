const urlAPI = "https://script.google.com/macros/s/AKfycbyJwFrDgGcJqe94vWLklyDqhby8MvSs_cN-iuQVP2cnHZzqnFdVlorbuQd1b6Bk1JkW/exec"; async function carregarDashboard() { const statusGeral = document.getElementById('barra-geral'); try { // Tentativa de busca simples const resposta = await fetch(urlAPI, { method: 'GET', mode: 'cors', // Google exige CORS para o JSON ser lido redirect: 'follow' }); if (!resposta.ok) throw new Error('Falha na resposta do servidor'); const dados = await resposta.json(); console.log("Dados AgroSB carregados:", dados); // Inserção no HTML document.getElementById('data-inicio').innerText = "📅 INÍCIO: " + new Date(dados.info.inicio).toLocaleDateString('pt-BR', {timeZone: 'UTC'}); document.getElementById('data-atualizacao').innerText = "🔄 ATUALIZAÇÃO: " + new Date(dados.info.atualizacao).toLocaleDateString('pt-BR', {timeZone: 'UTC'}); statusGeral.innerText = `Produção Total: ${dados.info.operacao.toLocaleString('pt-BR')} Metros`; // Renderizar Retiros (Alvorada, Modelo, etc) const container = document.getElementById('lista-retiros'); container.innerHTML = dados.retiros.map(r => `
${r.nome} ${r.executado} / ${r.total} m
${r.perc}%
`).join(''); } catch (erro) { console.error("Erro técnico:", erro); statusGeral.innerText = "Erro de conexão. Verifique o console (F12)."; } } // Chama a função ao carregar a página window.onload = carregarDashboard;
